\section{Vector Calculus}

For a scalar field $\phi$ and a vector field $\vect{A}$, the most common
differential operators are
\begin{equation}
  \grad \phi, \qquad
  \grad \cdot \vect{A}, \qquad
  \grad \times \vect{A}, \qquad
  \laplacian \phi.
  \label{eq:vector-operators}
\end{equation}

\begin{theorem}[Divergence theorem]
  If the boundary of a volume $V$ is the closed surface $\partial V$, then
  \begin{equation}
    \iiint_V (\grad\cdot\vect{A})\,\dd V
    = \oiint_{\partial V} \vect{A}\cdot\dd\vect{S}.
  \end{equation}
\end{theorem}
